## Global Constraints

- **基线 Bash 3.2**：不用 `local -n` / `declare -A` / 大写展开 / `mapfile` / `printf -v`（tools/ 下的门禁脚本是 bash，可用 bash 特性；`lib/posix/` 一律纯 POSIX sh）。
- **`lib/posix/` 只许 POSIX 语法**：无 `local`、无 `[[`、无 `declare`、无 `+=`、无 `((`；函数内自变量用 `_sp_` 前缀专属名。违反会被 `make posix-check` 拦下。
- **测试语言钉死 en**：`tests/test_helper.sh` 已 `export SHELLKIT_LANG=en`；断言断**完整文案**（坑 12）。
- **管道纪律**：不写 `grep -q` / `head` 打上游（坑 5，SIGPIPE 141）；截断用 `sed -n`。
- **变量声明一行一个**（坑 27）；赋值右边可能含通配符时整体加引号（坑 6）。
- **macOS BSD sed**：`sed -i` 必须带后缀，或「sed … > tmp && mv」形态；本计划所有 sed 用输出到临时文件的形态。
- **生成物不手改**：`capabilities.tsv`、`tools/api_baseline.txt`、`docs/API-FREEZE.md`、`docs/REFERENCE.md`、`AGENTS.md` 一律用对应命令重生成。
- **每个新判据先红后绿、带变异自证**（形状守卫：变异必须 cmp 证明落到源码上）。
- **提交纪律**：每个 Task 结束提交一次；提交信息按仓库风格（feat/ fix/ docs/ chore/）。
- **不碰 `tools/coverage_branch_baseline.txt`**：本计划不改任何被测分支的结构。

在 `lib/posix/shellkit-posix.sh` 中 `posix_capable() {` 定义**之前**插入：

```sh
# ---- 平台探测与前置检查（与主库同名同义）----

# SK_OS / SK_ARCH —— 与主库同名同义。主库用 \$OSTYPE 零 fork；设备侧没有可靠
# 的 OSTYPE 口径 ⇒ 装载时各 fork 一次 uname（只在 source 这份库时跑一次）。
_sk_posix_os_detect() {
  case "$(uname -s 2>/dev/null)" in
    linux* | Linux*) SK_OS=linux ;;
    darwin* | Darwin*) SK_OS=darwin ;;
    *) SK_OS=other ;;
  esac
  case "$(uname -m 2>/dev/null)" in
    arm64* | aarch64*) SK_ARCH=aarch64 ;;
    x86_64* | amd64*) SK_ARCH=x86_64 ;;
    *) SK_ARCH=other ;;
  esac
}
_sk_posix_os_detect

# os_name —— linux | darwin | other
os_name() { printf '%s\n' "$SK_OS"; }

# os_arch —— x86_64 | aarch64 | other
os_arch() { printf '%s\n' "$SK_ARCH"; }

# os_is_root —— 当前是否以 root 身份运行（id -u 为 0 ⇒ 0，否则 1）
os_is_root() {
  if [ "$(id -u 2>/dev/null)" = "0" ]; then
    return 0
  fi
  return 1
}

# os_has_cmd CMD... —— 零参数退 2；任一命令缺失退 1（与主库同口径，不记 SK_ERRMSG）
os_has_cmd() {
  if [ "$#" -lt 1 ]; then
    log_error "Usage: os_has_cmd CMD..."
    return 2
  fi
  for _sp_c in "$@"; do
    if ! command -v "$_sp_c" >/dev/null 2>&1; then
      return 1
    fi
  done
  return 0
}

# require_cmd CMD... —— 零参数退 2；缺命令退 3 并记 SK_ERRMSG（与主库同口径）
require_cmd() {
  if [ "$#" -lt 1 ]; then
    log_error "Usage: require_cmd CMD..."
    return 2
  fi
  for _sp_c in "$@"; do
    if ! command -v "$_sp_c" >/dev/null 2>&1; then
      log_error "Required command not found: $_sp_c"
      return 3
    fi
  done
  return 0
}
```

注意：无 `local`，循环变量 `_sp_c` 是专属前缀名；`SK_OS`/`SK_ARCH` 是与主库同名的公开变量（两份库不同时加载，与 `proc_*` 同一条既有约定）。

```bash
#!/usr/bin/env bash
# 能力清单的**理由口径**判据：新能力必须是 yes，旧的批量套话必须换掉。
# 为什么单独一条：清单是设备侧选型的依据 —— 理由写错（把 os_name 的缺席
# 归因于「包管理器」）会把人引向错误判断，而 posix-check 只查「有没有写理由」，
# 不查理由对不对。
. "$(cd "$(dirname "$0")" && pwd)/test_helper.sh"
ROOT="$TEST_ROOT"
CAP="$ROOT/lib/posix/capabilities.tsv"

# _field FILE FN COL —— 清单里 FN 的第 COL 列（2=有没有，3=理由/备注）
_field() {
  awk -F'\t' -v f="$2" -v c="$3" '$1==f{print $c}' "$1"
}

# _check_reasons FILE —— 全部理由断言；通过返回 0，违反返回 1。
# 独立成函数是为了变异自证：把源码的理由表改回套话、重新生成清单后，
# 同一批断言必须判红（否则断言是恒真的）。
_check_reasons() {
  local __cap=$1
  local __bad=0
  local __v
  local __fn
  # 1) 五个新能力必须是 yes
  for __fn in os_name os_arch os_is_root os_has_cmd require_cmd; do
    __v=$(_field "$__cap" "$__fn" 2)
    if [ "$__v" != "yes" ]; then
      printf 'FAIL: %s 应进 POSIX 子集，清单里是 <%s>\n' "$__fn" "$__v" >&2
      __bad=1
    fi
  done
  # 2) os_distro_id 的 no 理由不许再是「包管理器」套话（它是发行版探测）
  __v=$(_field "$__cap" os_distro_id 3)
  case "$__v" in
    '')
      printf 'FAIL: os_distro_id 没有理由\n' >&2
      __bad=1
      ;;
    *包管理器*)
      printf 'FAIL: os_distro_id 的理由仍是包管理器套话: %s\n' "$__v" >&2
      __bad=1
      ;;
  esac
  # 3) os_sha256 的理由必须指向 net_sha256（哈希已在子集里）
  __v=$(_field "$__cap" os_sha256 3)
  case "$__v" in
    *net_sha256*) ;;
    *)
      printf 'FAIL: os_sha256 的理由应指向 net_sha256: %s\n' "$__v" >&2
      __bad=1
      ;;
  esac
  # 4) os_pkg_install 保留真实理由（这条本来就对，钉住不许被顺手改坏）
  __v=$(_field "$__cap" os_pkg_install 3)
  case "$__v" in
    *系统包*) ;;
    *)
      printf 'FAIL: os_pkg_install 的理由应说明系统包: %s\n' "$__v" >&2
      __bad=1
      ;;
  esac
  return "$__bad"
}

# ---- 真清单必须过 ----
if ! _check_reasons "$CAP"; then
  _fail "能力清单的理由口径不达标（明细见上方 FAIL）"
fi

# ---- 变异自证：把 os.sh 模块默认理由改回套话 ⇒ 同一批断言必须判红 ----
d="$(tmpdir)"
mkdir -p "$d/tools" "$d/lib"
cp -R "$ROOT/lib/." "$d/lib/"
sed 's|POSIX 子集只给 os_name / os_arch / os_is_root / os_has_cmd（平台感知与命令前置检查）；系统包、桌面与发行版探测不进子集|设备侧没有包管理器，也不该动系统包|'   "$ROOT/tools/check_posix.sh" >"$d/tools/check_posix.sh"
if cmp -s "$ROOT/tools/check_posix.sh" "$d/tools/check_posix.sh"; then
  _fail "变异没落到源码上（os.sh 模块默认理由没变）—— 判据自己坏了"
fi
if ! SHELLKIT_CHECK_ROOT="$d" bash "$d/tools/check_posix.sh" --write >/dev/null 2>&1; then
  _fail "变异清单生成失败（check_posix --write 在夹具上退了非 0）"
fi
if _check_reasons "$d/lib/posix/capabilities.tsv" >/dev/null 2>&1; then
  _fail "变异自证失败：理由表改回套话后判据没变红 —— 断言是恒真的"
fi

echo "OK: 能力清单理由口径正确，变异自证通过"
```

- [ ] **Step 4: 实现门禁**

新建 `tools/check_deprecation.sh`（内容如下；若写入后发现反引号转义被工具吃掉，以 `bash -n` 与 Step 5 判据跑绿为准校对）：

```bash
#!/usr/bin/env bash
#
# 门禁：弃用承诺必须兑现（docs/COMPATIBILITY.md 的「弃用流程」）。
#
# 承诺三步：① 源码写 @deprecated <版本> 并同步 docs/MIGRATING.md；
# ② 保留至少一个已发布的 minor；③ 然后才允许删除。
# COMPATIBILITY 过去自认「靠迁移表与评审守，没有自动判据」—— 没有执法的
# 承诺随时可能被无声违反。本门禁查四件事：
#   a) 每个 @deprecated 注解版本合法，且 tools/deprecated.txt 有同版本台账；
#   b) 被弃用的函数在 docs/MIGRATING.md 里提得到（反引号包着的函数名）；
#   c) 台账登记了移除的：源码里必须真的没了、弃用到移除满一个 minor、
#      移除版本不晚于当前 VERSION、基线里留有记录、迁移表里提得到；
#   d) 反向：基线里有、源码里没有的函数，台账必须登记了移除版本。
#
# 空状态（零注解、台账只有头注释）是**合法**的：门禁已武装、没有目标。
# 判据能力由 tests/test_deprecation_gate.sh 双向 + 变异自证证明。
#
#   bash tools/check_deprecation.sh    # 0 通过；5 不合格；2 用法错
set -u

ROOT="${SHELLKIT_CHECK_ROOT:-$(cd "$(dirname "${BASH_SOURCE[0]}")/.." &>/dev/null && pwd)}"
MIGRATING="$ROOT/docs/MIGRATING.md"
BASELINE="$ROOT/tools/api_baseline.txt"
LEDGER="$ROOT/tools/deprecated.txt"
VER_FILE="$ROOT/VERSION"
SEMVER="$ROOT/tools/ver_semver.sh"

_bad=0
_fail() {
  printf 'deprecation-check: %s\n' "$1" >&2
  _bad=1
}

case "${1:-}" in
  "") ;;
  -h | --help)
    printf 'usage: bash %s\n' "${0##*/}"
    exit 0
    ;;
  *)
    printf 'deprecation-check: unknown option: %s\n' "$1" >&2
    exit 2
    ;;
esac

for __req in "$MIGRATING" "$BASELINE" "$LEDGER" "$VER_FILE" "$SEMVER"; do
  if [ ! -f "$__req" ]; then
    printf 'deprecation-check: missing %s\n' "$__req" >&2
    exit 5
  fi
done

# shellcheck source=tools/ver_semver.sh
. "$SEMVER"

_cur="$(tr -d '[:space:]' <"$VER_FILE")"
if ! printf '%s\n' "$_cur" | grep -Eq '^[0-9]+\.[0-9]+\.[0-9]+(-[0-9A-Za-z.]+)?$'; then
  printf 'deprecation-check: VERSION is not x.y.z: <%s>\n' "$_cur" >&2
  exit 5
fi

# _is_ver V —— x.y.z（可带预发布后缀）
_is_ver() {
  printf '%s\n' "$1" | grep -Eq '^[0-9]+\.[0-9]+\.[0-9]+(-[0-9A-Za-z.]+)?$'
}

# _ver_major / _ver_minor —— 取主 / 次版本号（纯数字）
_ver_major() {
  printf '%s' "${1%%.*}"
}
_ver_minor() {
  __mv=${1#*.}
  printf '%s' "${__mv%%.*}"
}

# _window_ok DEP CUR —— 从 DEP 到 CUR 至少满一个 minor
# （同 major 要求 minor 差 ≥1；major 更大也算过；major 更小不算）
_window_ok() {
  __dj=$(_ver_major "$1")
  __dm=$(_ver_minor "$1")
  __cj=$(_ver_major "$2")
  __cm=$(_ver_minor "$2")
  if [ "$__cj" -gt "$__dj" ]; then return 0; fi
  if [ "$__cj" -lt "$__dj" ]; then return 1; fi
  [ $((__cm - __dm)) -ge 1 ]
}

# _src_has FN —— 函数还在 lib/core 源码里吗
_src_has() {
  grep -Eq "^$1\(\)[ \t]*\{" "$ROOT"/lib/core/*.sh 2>/dev/null
}

WORK="$(mktemp -d "${TMPDIR:-/tmp}/shellkit-deprecation.XXXXXX")" || exit 3
trap 'rm -rf "$WORK"' EXIT
ANN="$WORK/ann.tsv"
BASE_NAMES="$WORK/base.list"

# ---- 源码里的 @deprecated 注解：函数名<TAB>版本 ----
# 注解行与定义之间允许夹说明注释；隔了非注释内容 ⇒ 悬空（单独报）。
awk '
  /^# @deprecated[ \t]+[0-9]/ { ver = $3; pending = 1; next }
  /^[A-Za-z_][A-Za-z0-9_]*\(\)[ \t]*\{/ && pending {
    name = $0
    sub(/\(.*/, "", name)
    print name "\t" ver
    pending = 0
    next
  }
  pending && /^#/ { next }
  pending { pending = 0 }
  END { if (pending) print "__DANGLING__\t" ver }
' "$ROOT"/lib/core/*.sh >"$ANN" || exit 5

# ---- 规则 a/b：注解侧 ----
while IFS=$'\t' read -r __fn __ver; do
  [ -n "$__fn" ] || continue
  if [ "$__fn" = "__DANGLING__" ]; then
    _fail "@deprecated 注解后面没有紧邻的函数定义（悬空注解）：$__ver"
    continue
  fi
  if ! _is_ver "$__ver"; then
    _fail "$__fn 的 @deprecated 版本不是 x.y.z: <$__ver>"
    continue
  fi
  if _ver_gt "$__ver" "$_cur"; then
    _fail "$__fn 的弃用版本 $__ver 晚于当前 VERSION $_cur（不能声明未来的弃用）"
  fi
  __led="$(awk -F'\t' -v f="$__fn" '$1==f{print $2}' "$LEDGER")"
  if [ -z "$__led" ]; then
    _fail "$__fn 有 @deprecated $__ver 注解，但 tools/deprecated.txt 没有它的台账行"
  elif [ "$__led" != "$__ver" ]; then
    _fail "$__fn 的注解版本 $__ver 与台账 $__led 不一致"
  fi
  if ! grep -qF "\`$__fn\`" "$MIGRATING"; then
    _fail "$__fn 有 @deprecated 注解，但 docs/MIGRATING.md 里没有反引号包着的 $__fn"
  fi
done <"$ANN"

# ---- 规则 a/c：台账侧 ----
while IFS=$'\t' read -r __fn __dep __rm; do
  case "$__fn" in '' | '#'*) continue ;; esac
  if ! _is_ver "$__dep"; then
    _fail "台账里 $__fn 的弃用版本不是 x.y.z: <${__dep:-}>"
    continue
  fi
  if _ver_gt "$__dep" "$_cur"; then
    _fail "台账里 $__fn 的弃用版本 $__dep 晚于当前 VERSION $_cur"
    continue
  fi
  if _src_has "$__fn"; then
    # 还在源码里：注解必须在（不许静默取消弃用），移除列必须是 -
    if ! awk -F'\t' -v f="$__fn" '$1==f{found=1} END{exit !found}' "$ANN"; then
      _fail "台账登记了 $__fn 被弃用，但源码里没有 @deprecated 注解（弃用被静默取消？）"
    fi
    if [ -n "${__rm:-}" ] && [ "$__rm" != "-" ]; then
      _fail "台账说 $__fn 已移除（$__rm），但源码里还有它的定义"
    fi
    continue
  fi
  # 已从源码删除：移除版本必须齐全且满窗口
  if [ -z "${__rm:-}" ] || [ "$__rm" = "-" ]; then
    _fail "台账里的 $__fn 已从源码删除，但第三列没有登记移除版本"
    continue
  fi
  if ! _is_ver "$__rm"; then
    _fail "台账里 $__fn 的移除版本不是 x.y.z: <$__rm>"
    continue
  fi
  if _ver_gt "$__rm" "$_cur"; then
    _fail "台账里 $__fn 的移除版本 $__rm 晚于当前 VERSION $_cur"
  fi
  if ! _window_ok "$__dep" "$__rm"; then
    _fail "台账里 $__fn 从弃用 $__dep 到移除 $__rm 不满一个 minor（COMPATIBILITY 承诺的窗口）"
  fi
  if ! awk -F'\t' -v f="$__fn" '$1==f{found=1} END{exit !found}' "$BASELINE"; then
    _fail "台账说 $__fn 被移除了，但 tools/api_baseline.txt 里没有它的记录（重生成基线？）"
  fi
  if ! grep -qF "\`$__fn\`" "$MIGRATING"; then
    _fail "台账里的 $__fn 在 docs/MIGRATING.md 里提不到（移除必须写迁移表）"
  fi
done <"$LEDGER"

# ---- 规则 d：反向 —— 基线里有、源码里没有 ⇒ 台账必须登记了移除 ----
awk -F'\t' '/^[A-Za-z_][A-Za-z0-9_]*\t/ {print $1}' "$BASELINE" >"$BASE_NAMES" || true
while IFS= read -r __fn; do
  [ -n "$__fn" ] || continue
  if _src_has "$__fn"; then continue; fi
  __rm="$(awk -F'\t' -v f="$__fn" '$1==f{print $3}' "$LEDGER")"
  if [ -z "$__rm" ] || [ "$__rm" = "-" ]; then
    _fail "$__fn 在基线里但源码里已经没有了 —— tools/deprecated.txt 必须登记它的移除版本"
  fi
done <"$BASE_NAMES"

if [ "$_bad" -ne 0 ]; then
  printf 'deprecation-check: FAILED\n' >&2
  exit 5
fi
printf 'deprecation-check: ok\n'
exit 0
```
